import datetime, hashlib, json, os, pathlib, shlex, subprocess, tempfile
utc = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
with tempfile.TemporaryDirectory(prefix='wfc-p3-h001-https-', dir=os.environ['TMPDIR']) as scratch:
    os.chmod(scratch, 0o700)
    for index, url in enumerate(('https://example.com/', 'https://www.iana.org/domains/reserved'), 1):
        fd, response = tempfile.mkstemp(prefix=f'response-{index}-', dir=scratch)
        os.close(fd)
        argv = ['/usr/bin/curl', '--disable', '--silent', '--show-error', '--fail',
                '--connect-timeout', '5', '--max-time', '20', '--proto', '=https',
                '--max-redirs', '0', '--output', response, '--write-out',
                'http_code=%{http_code}\nremote_ip=%{remote_ip}\nurl_effective=%{url_effective}\nssl_verify_result=%{ssl_verify_result}\nsize_download=%{size_download}\n', url]
        print('start_utc=' + utc(), flush=True)
        print('command=' + shlex.join(argv), flush=True)
        result = subprocess.run(argv, capture_output=True, text=True, timeout=25)
        print('end_utc=' + utc())
        print('exit_code=' + str(result.returncode))
        print('stdout=' + json.dumps(result.stdout[:2048]))
        print('stderr=' + json.dumps(result.stderr[:2048]))
        print('stdout_truncated=' + str(len(result.stdout) > 2048))
        print('stderr_truncated=' + str(len(result.stderr) > 2048))
        print('response_file_bytes=' + str(pathlib.Path(response).stat().st_size))
        if result.returncode == 0:
            print('response_sha256=' + hashlib.sha256(pathlib.Path(response).read_bytes()).hexdigest())
            break
        print('response_sha256=not-computed-request-failed')
print('private_scratch_removed=True')
